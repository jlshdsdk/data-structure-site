实战1 · vector + sort：数组排序与去重

🎬 对应课程：L5 数组 + L17 排序 ｜ 考试规则见 L0/L26

【题目描述】

输入n和n个整数（可能重复），输出去重后的升序序列。如 5 / 3 1 4 1 5 → 1 3 4 5。

STL选型思路：vector承接输入；去重两条路——①sort+unique+erase（O(nlogn)原地）；②set边插边去重（更短但常数大）。展示①，②作对照。

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cin >> n;
    vector<int> v(n);                        // 开n个空间
    for (int i = 0; i < n; i++) cin >> v[i];

    sort(v.begin(), v.end());                // ① 排序：使重复元素相邻

    v.erase(unique(v.begin(), v.end()),      // ② unique挤走相邻重复
            v.end());                        // ③ erase删逻辑尾之后残留

    for (int i = 0; i < (int)v.size(); i++)
        cout << v[i] << (i + 1 == (int)v.size() ? '\n' : ' ');
    return 0;
}
易错点：①unique必须先排序；②erase的区间是[unique返回值, end)；③size()是无符号，与int比较要强转防告警。

实战2 · stack：括号匹配 与 表达式求值（后缀 + 中缀）

🎬 对应课程：L4 表达式求值（后缀版） ｜ 考试规则见 L0/L26

【题目A：括号匹配】

判断只含 ()[]{} 的串是否全部匹配。"([{}])"✔，"([)]"✘。

选型思路：左括号进栈等待，右括号必须与栈顶配对——LIFO恰好对应括号嵌套。

#include <iostream>
#include <stack>
#include <string>
using namespace std;

bool isMatch(const string &s) {
    stack<char> st;
    for (int i = 0; i < (int)s.size(); i++) {
        if (s[i] == '(' || s[i] == '[' || s[i] == '{')
            st.push(s[i]);                   // 左括号进栈等待
        else {
            if (st.empty()) return false;    // 右括号无左可配
            char t = st.top();
            if ((s[i] == ')' && t != '(') || (s[i] == ']' && t != '[') ||
                (s[i] == '}' && t != '{')) return false;
            st.pop();                        // 配对成功弹出
        }
    }
    return st.empty();                       // 栈非空=左括号悬空
}

int main() {
    string s;
    while (cin >> s) cout << (isMatch(s) ? "matched" : "not matched") << endl;
    return 0;
}
【题目B：后缀表达式求值】

输入后缀表达式（操作数为非负整数，运算符+ - * /，空格分隔），求值。如 "6 3 2 - *" → 6×(3-2)=6。408经典：后缀求值只需一个操作数栈。

#include <iostream>
#include <stack>
#include <sstream>
#include <string>
using namespace std;

int evaluatePostfix(const string &expr) {
    stack<int> num;                          // 操作数栈
    istringstream in(expr);
    string tok;
    while (in >> tok) {
        if (isdigit(tok[0])) {               // 操作数：转int压栈
            int v = 0;
            for (int i = 0; i < (int)tok.size(); i++)
                v = v * 10 + (tok[i] - '0');
            num.push(v);
        } else {                             // 运算符：弹两个数
            int b = num.top(); num.pop();    // 先弹的是右操作数！
            int a = num.top(); num.pop();
            int r = 0;
            if (tok == "+") r = a + b;
            if (tok == "-") r = a - b;       // a-b顺序不能反
            if (tok == "*") r = a * b;
            if (tok == "/") r = a / b;
            num.push(r);
        }
    }
    return num.top();
}

int main() {
    string s;
    while (getline(cin, s))
        cout << evaluatePostfix(s) << endl;  // 输入示例：6 3 2 - *
    return 0;
}
【题目C：中缀表达式求值（算符优先 · 双栈法）】

输入不含空格的 + − * / 与括号的中缀表达式（操作数为非负整数），求值。如 3+4*2-(1+2) → 8。与题目B合成408"栈的应用"两大表达式考点。

选型思路：经典双栈法——操作数栈 + 运算符栈；当前运算符优先级 ≤ 栈顶时先算栈内（保证先乘除后加减），遇 ) 清算到 ( 。

#include <iostream>
#include <stack>
#include <string>
using namespace std;

int prec(char op) {                          // 运算符优先级
    if (op == '*' || op == '/') return 2;
    if (op == '+' || op == '-') return 1;
    return 0;                                // '('入栈后按0，绝不会被弹出
}

void calc(stack<int> &num, stack<char> &ops) {   // 弹两数一符，算完压回
    int b = num.top(); num.pop();            // 注意：先弹的是右操作数
    int a = num.top(); num.pop();
    char op = ops.top(); ops.pop();
    int r = 0;
    if (op == '+') r = a + b;
    if (op == '-') r = a - b;                // a-b 顺序不能反！
    if (op == '*') r = a * b;
    if (op == '/') r = a / b;
    num.push(r);
}

int evaluate(const string &s) {
    stack<int> num;                          // 操作数栈
    stack<char> ops;                         // 运算符栈
    int i = 0, n = s.size();
    while (i < n) {
        if (isdigit(s[i])) {                 // 数字：完整读出
            int v = 0;
            while (i < n && isdigit(s[i]))
                v = v * 10 + (s[i++] - '0');
            num.push(v);
        } else if (s[i] == '(') {
            ops.push(s[i++]);                // 左括号直接进栈
        } else if (s[i] == ')') {
            while (ops.top() != '(')         // 算到左括号为止
                calc(num, ops);
            ops.pop();                       // 弹出'('本身
            i++;
        } else {                             // 四则运算符
            while (!ops.empty() && prec(ops.top()) >= prec(s[i]))
                calc(num, ops);              // 栈顶优先级不低于当前：先算
            ops.push(s[i++]);
        }
    }
    while (!ops.empty()) calc(num, ops);     // 清空剩余运算符
    return num.top();
}

int main() {
    string s;
    while (cin >> s)
        cout << evaluate(s) << endl;         // 输入示例：3+4*2-(1+2) → 8
    return 0;
}
双栈法判定句："新运算符优先级 ≤ 栈顶则先弹算"（左结合的体现）；笔试常考"中缀转后缀"手算——与本题同一套优先级规则，后缀求值则只需一个操作数栈（见题目B）。
易错点：①后缀求值只弹一个栈（中缀才要双栈）；②减法/除法先弹的是右操作数；③中缀转后缀手算（408选择题）与本题同一套优先级规则。

实战3 · queue + BFS：二叉树层序遍历 与 图的广度优先遍历

🎬 对应课程：L8 层次遍历 + L13 BFS ｜ 考试规则见 L0/L26

【题目A：层序遍历】

按层从左到右输出二叉树。输入先序序列（#表空），如 AB#D##C##。

#include <iostream>
#include <queue>
#include <string>
using namespace std;

typedef struct BiTNode {
    char data;
    struct BiTNode *lchild, *rchild;
} BiTNode, *BiTree;

BiTree build(const string &s, int &i) {      // 先序+#建树
    if (i >= (int)s.size() || s[i] == '#') { i++; return NULL; }
    BiTree T = new BiTNode;
    T->data = s[i++];
    T->lchild = build(s, i);
    T->rchild = build(s, i);
    return T;
}

void levelOrder(BiTree T) {
    if (!T) return;
    queue<BiTree> q;
    q.push(T);                               // 根入队
    while (!q.empty()) {
        BiTree p = q.front(); q.pop();       // 出队访问
        cout << p->data;
        if (p->lchild) q.push(p->lchild);    // 孩子依次入队
        if (p->rchild) q.push(p->rchild);
    }
    cout << endl;
}

int main() {
    string s; cin >> s;
    int i = 0;
    levelOrder(build(s, i));                 // AB#D##C## → ABCD
    return 0;
}
【题目B：无权图BFS最短距离】

n点m边无向图，求顶点1到各点最少步数，不可达输出-1。

#include <iostream>
#include <queue>
#include <vector>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> g(n + 1);            // vector邻接表
    for (int i = 0; i < m; i++) {
        int u, v; cin >> u >> v;
        g[u].push_back(v);                   // 无向图存两遍
        g[v].push_back(u);
    }
    vector<int> d(n + 1, -1);                // -1兼做未访问标记
    queue<int> q;
    d[1] = 0; q.push(1);
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : g[u])
            if (d[v] == -1) {                // 首次到达即最短
                d[v] = d[u] + 1;
                q.push(v);
            }
    }
    for (int i = 1; i <= n; i++) cout << d[i] << " ";
    cout << endl;
    return 0;
}
易错点：①图BFS要入队时标记（树不用，因为无环）；②分层处理（求宽度/右视图）用"rear-front=当前层结点数"；③FIFO保证"先入队的孩子先被访问"=层的定义。

实战4 · priority_queue：TopK问题 与 哈夫曼编码

🎬 对应课程：L11 哈夫曼WPL + L18 TopK ｜ 考试规则见 L0/L26

【题目A：TopK】

n个整数（n很大）求最大的k个，不允许全排序。

选型思路：维护大小为k的小根堆：堆顶是当前k个候选的最小者（候补席），新元素比堆顶大才替换。O(nlogk)。

#include <iostream>
#include <queue>
#include <vector>
#include <functional>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    priority_queue<int, vector<int>, greater<int>> q;   // 小根堆：堆顶最小
    for (int i = 0; i < n; i++) {
        int x; cin >> x;
        if ((int)q.size() < k) q.push(x);    // 堆未满直接放
        else if (x > q.top()) {              // 比当前第k大才有资格
            q.pop();                         // 换掉候选中最小的
            q.push(x);
        }
    }
    while (!q.empty()) {                     // 弹出即前k大（升序）
        cout << q.top() << " ";
        q.pop();
    }
    cout << endl;
    return 0;
}
【题目B：哈夫曼编码】

n个字符的权值，构造哈夫曼树并输出每个字符的01编码，以及WPL。

选型思路：小根堆完成"取最小两个合并"O(logn)；再DFS一遍树给叶子发编码（左0右1）。用 pair<权值, 结点编号> 入堆保证次序。

#include <iostream>
#include <queue>
#include <vector>
#include <string>
#include <functional>
using namespace std;

int lc[300], rc[300];                        // 左右孩子（静态树）
int w[300];

void dfs(int u, string code) {               // 给叶子发编码
    if (lc[u] == -1 && rc[u] == -1) {        // 叶子：输出编码
        cout << "w=" << w[u] << " code=" << code << endl;
        return;
    }
    dfs(lc[u], code + "0");                  // 左0
    dfs(rc[u], code + "1");                  // 右1
}

int main() {
    int n;
    cin >> n;
    priority_queue< pair<long long,int>, vector<pair<long long,int>>,
                    greater<pair<long long,int>> > q;   // 小根堆
    for (int i = 0; i < n; i++) {
        cin >> w[i];
        lc[i] = rc[i] = -1;                  // 叶子
        q.push({w[i], i});                   // (权值,编号)入堆
    }
    int node = n;                            // 新结点编号从n起
    long long wpl = 0;
    while ((int)q.size() > 1) {
        pair<long long,int> a = q.top(); q.pop();   // 最小
        pair<long long,int> b = q.top(); q.pop();   // 次小
        w[node] = w[a.second] + w[b.second];
        lc[node] = a.second; rc[node] = b.second;
        wpl += w[node];                      // 非叶权和=WPL
        q.push({w[node], node});
        node++;
    }
    int root = q.top().second;
    if (n == 1) cout << "w=" << w[0] << " code=0" << endl;  // 特判单结点
    else dfs(root, "");
    cout << "WPL = " << wpl << endl;
    return 0;
}
易错点：①"求最大k个用小根堆"——方向最易反；②WPL=非叶结点权值和，一行累加；③n=1时哈夫曼树退化，编码约定为"0"；④笔试版（静态数组两次扫最小）见第四部分3.5，两者必须都会。

实战5 · unordered_map：哈希统计 与 字符串查重

🎬 对应课程：L25 散列思想 ｜ 考试规则见 L0/L26

【题目A：两数之和】

n个整数找两数之和为target，输出下标（恰有一解）。

选型思路：扫到x时O(1)判定 target−x 是否出现过——空间换时间的教科书案例。

#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

int main() {
    int n, target;
    cin >> n >> target;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    unordered_map<int, int> pos;             // 值 → 下标
    for (int i = 0; i < n; i++) {
        int need = target - a[i];            // 还需要的另一半
        unordered_map<int,int>::iterator it = pos.find(need);
        if (it != pos.end()) {
            cout << it->second << " " << i << endl;
            return 0;
        }
        pos[a[i]] = i;                       // 边扫边登记
    }
    return 0;
}
【题目B：字符串查重+词频】

读入若干单词，统计各单词出现次数并输出（按出现次数降序、同次数字典序）。

选型思路：统计用unordered_map（O(1)/次）；有序输出必须再排序——把键值对搬进vector<pair>用sort，体现"unordered统计 + sort输出"组合拳。

#include <iostream>
#include <string>
#include <vector>
#include <unordered_map>
#include <algorithm>
using namespace std;

int main() {
    unordered_map<string, int> cnt;
    string w;
    while (cin >> w) cnt[w]++;               // []默认0再自增

    vector< pair<string,int> > v(cnt.begin(), cnt.end());
    sort(v.begin(), v.end(),
         [](const pair<string,int> &a, const pair<string,int> &b) {
             if (a.second != b.second) return a.second > b.second;  // 次数降序
             return a.first < b.first;                              // 字典序
         });
    for (int i = 0; i < (int)v.size(); i++)
        cout << v[i].first << " " << v[i].second << endl;
    return 0;
}
易错点：①cnt[w]++利用operator[]默认值——这里正是要用副作用；②判存在用find不用[]；③unordered遍历顺序不可依赖，有序输出必须排序。

实战6 · set / map：有序统计 与 区间查找

🎬 对应课程：L22/L23 有序查找 ｜ 考试规则见 L0/L26

【题目A：动态集合前驱后继】

维护集合，支持：1 x插入；2 x删除一个x；3 x查前驱（<x的最大值）；4 x查后继（>x的最小值）。单步必须O(logn)。

选型思路：unordered_set没有lower_bound——有序性不可替代，必须set + lower_bound/upper_bound + prev/next。

#include <iostream>
#include <set>
using namespace std;

int main() {
    int m;
    cin >> m;
    set<int> s;
    while (m--) {
        int op, x;
        cin >> op >> x;
        if (op == 1) s.insert(x);
        else if (op == 2) s.erase(s.find(x));        // 只删一个！
        else if (op == 3) {                          // 前驱
            set<int>::iterator it = s.lower_bound(x);
            if (it == s.begin()) cout << -1 << endl; // 无前驱
            else cout << *prev(it) << endl;
        } else {                                     // 后继
            set<int>::iterator it = s.upper_bound(x);
            if (it == s.end()) cout << -1 << endl;
            else cout << *it << endl;
        }
    }
    return 0;
}
【题目B：map有序统计+区间人数】

分数升序输出"分数:人数"，并查询[l,r]段总人数。

map<int, int> m;
for (int i = 0; i < n; i++) { int s; cin >> s; m[s]++; }
for (map<int,int>::iterator it = m.begin(); it != m.end(); ++it)
    cout << it->first << ":" << it->second << endl;
int total = 0;
for (map<int,int>::iterator it = m.lower_bound(l);      // 从≥l起
     it != m.end() && it->first <= r; ++it)             // 到>r停

易错点：①multiset删除单个要 erase(find(x))；②前驱=lower_bound的前一位（注意begin判界）；③区间统计 [l,r] 用 [lower_bound(l), upper_bound(r)) 半开区间。

实战7 · 单调队列/单调栈 STL实现：滑动窗口最大值 与 接雨水

🎬 对应课程：L18 堆 + L5 尺取法 ｜ 考试规则见 L0/L26

【题目A：滑动窗口最大值】

n个数、窗口k，输出每个窗口最大值。n≤10⁶，暴力O(nk)必超时。

选型思路：deque存下标、对应值递减：队尾出（淘汰小值）、队头出（过期）、队头即答案。

#include <iostream>
#include <vector>
#include <deque>
using namespace std;

int main() {
    int n, k;
    cin >> n >> k;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];

    deque<int> q;                            // 存下标：值从队头到队尾递减
    for (int i = 0; i < n; i++) {
        while (!q.empty() && a[q.back()] <= a[i])
            q.pop_back();                    // ①队尾出：≤当前者永无出头之日
        q.push_back(i);                      // ②当前下标入队尾
        if (q.front() <= i - k) q.pop_front();   // ③队头过期
        if (i >= k - 1) cout << a[q.front()] << " ";   // ④队头即最大
    }
    cout << endl;
    return 0;
}
【题目B：接雨水】

n根柱子宽度1，求能接多少雨水。三种解法中单调栈是408风格最自然的。

选型思路（单调栈版）：栈存下标、高度递减；来了更高的柱子，弹栈顶作"凹槽底"，左边界=新栈顶、右边界=当前柱，按"宽×高"累加。

#include <iostream>
#include <stack>
#include <vector>
using namespace std;

int trap(const vector<int> &h) {
    stack<int> st;                           // 栈存下标，高度递减
    long long water = 0;
    for (int i = 0; i < (int)h.size(); i++) {
        while (!st.empty() && h[i] > h[st.top()]) {
            int bottom = st.top(); st.pop(); // 凹槽底
            if (st.empty()) break;           // 无左边界，接不了
            int left = st.top();             // 左边界
            int width = i - left - 1;                        // 横向宽
            int height = min(h[left], h[i]) - h[bottom];     // 纵向高
            water += (long long)width * height;
        }
        st.push(i);
    }
    return (int)water;
}

int main() {
    int n;
    cin >> n;
    vector<int> h(n);
    for (int i = 0; i < n; i++) cin >> h[i];
    cout << trap(h) << endl;                 // 输入 0,1,0,2,1,0,1,3,2,1,2,1 → 6
    return 0;
}
易错点：①单调队列"先过期再取答案"顺序铁律；②接雨水弹栈后必须判栈空（无左边界）；③高度是 min(左右边界)−底，宽度按横向层算；④LC42也可用双指针O(1)空间解（见第七部分）。

实战8 · vector邻接表 + Dijkstra / 拓扑排序

🎬 对应课程：L12 存储 + L14 拓扑 + L15 Dijkstra ｜ 考试规则见 L0/L26

【题目A：堆优化Dijkstra】

n点m边的带权有向图，求1到各点最短距离（边权非负，可能有重边自环，不可达输出-1）。

选型思路：vector邻接表存 pair<邻点,边权>；小根堆存 pair<dist,顶点>，堆顶即"当前最小的未确认dist"，替代教材版的O(V)扫描。

#include <iostream>
#include <vector>
#include <queue>
#include <cstring>
using namespace std;
typedef long long ll;
typedef pair<ll,int> P;                      // (dist, 顶点)

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<P>> g(n + 1);              // g[u] = {(v, w), ...}
    for (int i = 0; i < m; i++) {
        int u, v; ll w;
        cin >> u >> v >> w;
        g[u].push_back({v, w});              // 有向图存一遍；无向存两遍
    }
    vector<ll> dist(n + 1, -1);
    vector<bool> done(n + 1, false);
    priority_queue<P, vector<P>, greater<P>> q;   // 小根堆：dist小在前
    dist[1] = 0;
    q.push({0, 1});
    while (!q.empty()) {
        P t = q.top(); q.pop();
        int u = t.second;
        if (done[u]) continue;               // 旧版本入堆，跳过
        done[u] = true;
        for (int i = 0; i < (int)g[u].size(); i++) {
            int v = g[u][i].first;
            ll nd = dist[u] + g[u][i].second;                 // 松弛
            if (dist[v] == -1 || nd < dist[v]) {
                dist[v] = nd;
                q.push({nd, v});             // 允许重复入堆，出堆时判done
            }
        }
    }
    for (int i = 1; i <= n; i++) cout << dist[i] << " ";
    cout << endl;
    return 0;
}
【题目B：拓扑排序】

n门课m个先修关系，输出一个可行修课顺序；有环输出"impossible"。

#include <iostream>
#include <vector>
#include <queue>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> g(n + 1);
    vector<int> indeg(n + 1, 0);
    for (int i = 0; i < m; i++) {
        int u, v; cin >> u >> v;             // 先修u，后修v
        g[u].push_back(v);
        indeg[v]++;
    }
    queue<int> q;
    for (int i = 1; i <= n; i++)
        if (indeg[i] == 0) q.push(i);        // 入度0进队
    int cnt = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        cout << u << " ";
        cnt++;
        for (int v : g[u])
            if (--indeg[v] == 0) q.push(v);  // 删边，新0入度入队
    }
    if (cnt < n) cout << "\nimpossible" << endl;   // 输出不全→有环
    return 0;
}
易错点：①堆优化Dijkstra用"懒删除"：同一顶点可能多次入堆，出堆时判done跳过；②dist用long long防溢出；③拓扑排序输出数<n⇔有环——这是判环标准方法；④笔试版（邻接矩阵O(V²)）见第四部分4.4/4.3。

实战9 · lower_bound 二分查找进阶应用

🎬 对应课程：L22 对半查找变体（BinSearchFirstPosGEK） ｜ 考试规则见 L0/L26

【题目A：区间统计】

非降序数组a[0..n-1]，q次询问：统计值在 [l, r] 内的元素个数。

选型思路：每次询问O(logn)：个数 = upper_bound(r) − lower_bound(l)，两次二分而非逐个比较。

#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, q;
    cin >> n >> q;
    int a[100005];
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);                          // 保证非降序（lower_bound前提）
    while (q--) {
        int l, r;
        cin >> l >> r;
        int cnt = upper_bound(a, a + n, r) - lower_bound(a, a + n, l);
        cout << cnt << endl;                 // [l,r]内元素个数
    }
    return 0;
}
【题目B：最长递增子序列 LIS（贪心+二分）】

求数组最长严格递增子序列长度。DP是O(n²)，贪心+二分O(nlogn)。

选型思路：维护数组tails——tails[i]=长度i+1的递增子序列的最小结尾（恒有序）；每个x用lower_bound找第一个≥x的位置并替换/追加——替换保证结尾尽量小，给未来留更多机会。

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int lengthOfLIS(const vector<int> &a) {
    vector<int> tails;                       // tails[i]：长度i+1的LIS最小结尾
    for (int i = 0; i < (int)a.size(); i++) {
        int x = a[i];
        vector<int>::iterator it = lower_bound(tails.begin(), tails.end(), x);
        if (it == tails.end()) tails.push_back(x);   // x比所有结尾大：LIS延长
        else *it = x;                        // 否则替换第一个≥x的结尾
    }
    return (int)tails.size();
}

int main() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    cout << lengthOfLIS(a) << endl;          // 10,9,2,5,3,7,101,18 → 4
    return 0;
}
易错点：①lower_bound前提必须有序——题目A要先sort，B的tails天然有序；②"找第一个≥x"用lower_bound（严格上升LIS），"找第一个>x"用upper_bound（非递减LIS变体）；③tails数组不是实际的LIS，只有长度有意义。
