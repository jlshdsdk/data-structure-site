LC 283移动零简单必考模板
考点拆解｜数据结构：数组 ｜ STL：swap ｜ 算法：同向双指针（读写指针） ｜ 突破口：k为已保留区边界，非零前移

把数组中所有0移到末尾，保持非零元素相对顺序。要求原地、尽量少操作。

查看解析

思路推导："保持非零相对顺序+原地"→读写双指针：k指向下一个非零应放的位置，i扫描，非零就交换到k。一趟O(n)。

class Solution {
public:
    void moveZeroes(vector<int>& nums) {
        int k = 0;                           // k：下一个非零元素的位置
        for (int i = 0; i < (int)nums.size(); i++) {
            if (nums[i] != 0) {
                swap(nums[i], nums[k]);      // 非零前移，零自然沉底
                k++;
            }
        }
    }
};

LC 88合并两个有序数组简单高频考点
考点拆解｜数据结构：数组 ｜ STL：无 ｜ 算法：逆向双指针（归并变体） ｜ 突破口：从后往前合并，避免覆盖nums1元素

nums1末尾预留了nums2长度的空位，将nums2合并入nums1使整体非降序。

查看解析

思路推导：正向归并会覆盖nums1的有效数据；预留空位在尾部→从后往前比较取大者放尾部。这是归并排序Merge的"镜像写法"，408归并变体高频。

class Solution {
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
        int i = m - 1, j = n - 1;            // 分别指向两数组末尾有效元素
        int k = m + n - 1;                   // 合并结果末尾
        while (i >= 0 && j >= 0) {
            if (nums1[i] > nums2[j]) nums1[k--] = nums1[i--];
            else                     nums1[k--] = nums2[j--];
        }
        while (j >= 0) nums1[k--] = nums2[j--];   // nums2剩余（nums1剩余本来就在）
    }
};

LC 121买卖股票的最佳时机简单
考点拆解｜数据结构：数组 ｜ STL：min ｜ 算法：一次扫描维护前缀最小值 ｜ 突破口：第i天卖出的最优=当天价−之前最低价

已知每天股价，某天买入、之后某天卖出，求最大利润；不行返回0。

查看解析

思路推导：枚举卖点i，买点必然是前i-1天最低价→扫描中维护minPrice，每步用 price−minPrice 更新答案。O(n)/O(1)。

class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int minPrice = prices[0];            // 到目前为止的最低价
        int best = 0;
        for (int i = 1; i < (int)prices.size(); i++) {
            best = max(best, prices[i] - minPrice);   // 今天卖
            minPrice = min(minPrice, prices[i]);      // 更新前缀最小
        }
        return best;
    }
};

查看解析

思路推导：dp[i]=以i结尾的最大子数组和：dp[i]=max(nums[i], dp[i-1]+nums[i])——前缀和为负则"拖后腿"，从i重新开始。滚动变量压成一维。

class Solution {

            cur = max(nums[i], cur + nums[i]);   // 接上前缀 or 重新开始
            best = max(best, cur);
        }
        return best;
    }
};

class Solution {
public:

        int n = nums.size();

        reverse(nums.begin(), nums.end());       // ①整体逆置
        reverse(nums.begin(), nums.begin() + k); // ②逆前k个
        reverse(nums.begin() + k, nums.end());   // ③逆剩余
    }
};

LC 560和为K的子数组中等高频考点
考点拆解｜数据结构：数组+哈希 ｜ STL：unordered_map ｜ 算法：前缀和 ｜ 突破口：pre[j]−pre[i]=k ⇔ 数pre[j]−k出现次数

统计和恰为k的连续子数组个数（含负数，滑动窗口失效）。

查看解析

思路推导：含负数→窗口单调性失效→前缀和：以j结尾的和为k的子数组个数=此前 pre−k 出现的次数，用哈希表边扫边计数。注意先记pre[0]=1（空前缀）。

class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int, int> cnt;         // 前缀和 → 出现次数
        cnt[0] = 1;                          // 空前缀：和为0出现1次
        int pre = 0, ans = 0;
        for (int x : nums) {
            pre += x;
            unordered_map<int,int>::iterator it = cnt.find(pre - k);
            if (it != cnt.end()) ans += it->second;   // 以当前结尾的和为k
            cnt[pre]++;
        }
        return ans;
    }
};

LC 11盛最多水的容器中等必考模板
考点拆解｜数据结构：数组 ｜ STL：min/max ｜ 算法：相向双指针+贪心 ｜ 突破口：移动较矮的一侧才可能更优

n条垂线与x轴组成容器，选两条使盛水最多：面积=min(h[i],h[j])×距离。

查看解析

思路推导：l、r两端开始，面积由较矮者决定；固定较矮者再怎么移都不可能更大→必须移动较矮一侧。每步O(1)淘汰一列，O(n)。

class Solution {
public:
    int maxArea(vector<int>& h) {
        int l = 0, r = (int)h.size() - 1, best = 0;
        while (l < r) {
            int area = min(h[l], h[r]) * (r - l);
            best = max(best, area);
            if (h[l] < h[r]) l++;            // 矮的一侧移动
            else             r--;
        }
        return best;
    }
};

📌 第四部分·1.3 双指针模板②｜与LC15三数之和同范式

LC 15三数之和中等高频考点
考点拆解｜数据结构：数组 ｜ STL：sort ｜ 算法：排序+枚举一个+首尾双指针 ｜ 突破口：排序后"两数之和"线性扫；三重去重

找出所有不重复的三元组使三数之和为0。

查看解析

思路推导：暴力O(n³)→排序后固定i，对i+1..n-1做首尾双指针找两数和=−nums[i]；三处去重：i跳过重复、命中后l/r各自跳过重复。O(n²)。

class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        vector<vector<int>> res;
        sort(nums.begin(), nums.end());      // ①排序：双指针+去重前提
        for (int i = 0; i < (int)nums.size(); i++) {
            if (nums[i] > 0) break;                       // 最小数>0无解
            if (i > 0 && nums[i] == nums[i-1]) continue;  // ②i去重
            int l = i + 1, r = (int)nums.size() - 1;
            while (l < r) {
                int s = nums[i] + nums[l] + nums[r];
                if (s < 0) l++;
                else if (s > 0) r--;
                else {
                    res.push_back({nums[i], nums[l], nums[r]});
                    while (l < r && nums[l] == nums[l+1]) l++; // ③去重
                    while (l < r && nums[r] == nums[r-1]) r--;
                    l++; r--;
                }
            }
        }
        return res;
    }
};

📌 第四部分·1.3 双指针模板②｜第三部分·1.10 sort

LC 287寻找重复数中等真题对应
考点拆解｜数据结构：数组(视作隐式链表) ｜ STL：无 ｜ 算法：快慢指针(Floyd判圈) ｜ 突破口：i→nums[i]成链，重复数=环入口

n+1个数取值1..n，至少一个重复。不修改数组、O(1)空间找重复数。

查看解析

思路推导：把数组看成链表 i→nums[i]，因值域1..n必有环，重复数=环的入口→快慢指针判圈+找入口（与第四部分快慢指针同款）。二分"≤mid的个数"也可解。

class Solution {
public:
    int findDuplicate(vector<int>& nums) {
        int slow = 0, fast = 0;
        do {
            slow = nums[slow];               // 慢针走1步
            fast = nums[nums[fast]];         // 快针走2步
        } while (slow != fast);              // 相遇：在环内
        slow = 0;
        while (slow != fast) {               // 双针同速，再次相遇=环入口
            slow = nums[slow];
            fast = nums[fast];
        }
        return slow;                         // 重复数
    }
};

📌 第四部分·1.4 快慢指针｜判圈思想同LC141/LC142

LC 42接雨水困难高频考点
考点拆解｜数据结构：数组/栈 ｜ STL：stack ｜ 算法：双指针 或 单调栈 ｜ 突破口：每列水位=min(左最大,右最大)−自身

柱状图接雨水总量。（第六部分·实战e7有单调栈版，此处给O(1)空间双指针版）

查看解析

思路推导：water[i]=min(左边最高,右边最高)−h[i]。双指针+两侧最大值：较矮一侧的答案已确定（min由它决定），逐列累加，O(n)/O(1)。

class Solution {
public:
    int trap(vector<int>& h) {
        int l = 0, r = (int)h.size() - 1;
        int lmax = 0, rmax = 0, water = 0;
        while (l < r) {
            lmax = max(lmax, h[l]);          // 左侧最高
            rmax = max(rmax, h[r]);          // 右侧最高
            if (lmax < rmax) water += lmax - h[l++];   // 较矮一侧已确定
            else             water += rmax - h[r--];
        }
        return water;
    }
};

class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode *prev = NULL, *cur = head;
        while (cur) {
            ListNode *next = cur->next;  // ①暂存后继防断链
            cur->next = prev;            // ②反转指向
            prev = cur;                  // ③prev前移
            cur = next;                  // ④cur前移
        }
        return prev;                     // 新表头
    }
};

LC 21合并两个有序链表简单必考模板
考点拆解｜数据结构：单链表 ｜ STL：无 ｜ 算法：双指针尾插归并 ｜ 突破口：虚拟头结点统一"插到表头"逻辑

合并两个非降序链表为一个非降序链表。

查看解析

思路推导：归并排序Merge的链表版：虚拟头结点+尾指针r，取小者尾插；剩余整段接上。

class Solution {
public:
    ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
        ListNode dummy(0);                   // 虚拟头结点（免特判第一个结点）
        ListNode *r = &dummy;                // 尾指针
        while (l1 && l2) {
            if (l1->val <= l2->val) { r->next = l1; l1 = l1->next; }
            else                    { r->next = l2; l2 = l2->next; }
            r = r->next;
        }
        r->next = l1 ? l1 : l2;              // 整段接管剩余
        return dummy.next;
    }
};

LC 141环形链表简单必考模板
考点拆解｜数据结构：单链表 ｜ STL：无 ｜ 算法：快慢指针Floyd判圈 ｜ 突破口：快2慢1，有环必相遇

判断链表是否有环。要求O(1)空间。

查看解析

思路推导：有环则快针"套圈"追上慢针；无环快针先到NULL。哈希表存地址也可解但O(n)空间——面试/机试追问点。

class Solution {
public:
    bool hasCycle(ListNode *head) {
        ListNode *slow = head, *fast = head;
        while (fast && fast->next) {
            slow = slow->next;               // 慢1步
            fast = fast->next->next;         // 快2步
            if (slow == fast) return true;   // 相遇有环
        }
        return false;                        // 到NULL无环
    }
};

📌 第四部分·1.4 快慢指针模板②｜进阶：LC142 找环入口

LC 142环形链表 II中等
考点拆解｜数据结构：单链表 ｜ STL：无 ｜ 算法：Floyd判圈+数学推导找入口 ｜ 突破口：相遇后一头回起点，同速再走再相遇即入口

返回链表入环的第一个结点；无环返回NULL。

查看解析

思路推导：设头到入口长a、相遇点在环内。相遇时慢针走了a+b，快针走了a+b+nb圈⇒a+b=nb(快针速度2倍)⇒a=nb−b=c(n-1)+... 结论：从相遇点和表头同时出发同速前进，再次相遇处即入口。

class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        ListNode *slow = head, *fast = head;
        while (fast && fast->next) {
            slow = slow->next;
            fast = fast->next->next;
            if (slow == fast) {              // 相遇
                slow = head;                 // 一针回头
                while (slow != fast) {       // 同速前进
                    slow = slow->next;
                    fast = fast->next;
                }
                return slow;                 // 环入口
            }
        }
        return NULL;
    }
};

LC 19删除链表的倒数第 N 个结点中等必考模板
考点拆解｜数据结构：单链表 ｜ STL：无 ｜ 算法：快慢指针+虚拟头结点 ｜ 突破口：fast先走N步，删除需停留在前驱

一趟扫描删除倒数第n个结点。

查看解析

思路推导：fast先走n步，slow/fast同速前进到fast为空，slow即倒数第n+1个（前驱）→改指针删除。虚拟头结点解决"删头"特判。

class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode dummy(0);
        dummy.next = head;
        ListNode *fast = &dummy, *slow = &dummy;
        for (int i = 0; i < n; i++) fast = fast->next;   // fast先走n步
        while (fast->next) {                 // 同速走到尾部
            fast = fast->next;
            slow = slow->next;
        }
        ListNode *del = slow->next;          // slow是前驱
        slow->next = del->next;
        delete del;
        return dummy.next;
    }
};

LC 160相交链表简单
考点拆解｜数据结构：单链表 ｜ STL：无 ｜ 算法：双指针"走完换道" ｜ 突破口：a+c+b == b+c+a 消除长度差

找两单链表第一个公共结点；不相交返回NULL。（408真题原题思想）

查看解析

思路推导：两针各从自己表头走，到尾换到对方表头——走过的总长相同（a+c+b 与 b+c+a），必在交点相遇；不相交则同到NULL。O(m+n)/O(1)。笔试版：先求两表长度差，长表先走差值步再同步走。

class Solution {
public:
    ListNode *getIntersectionNode(ListNode *headA, ListNode *headB) {
        ListNode *p = headA, *q = headB;
        while (p != q) {
            p = p ? p->next : headB;         // 走完A换B
            q = q ? q->next : headA;         // 走完B换A
        }
        return p;                            // 相交点或NULL
    }
};

📌 第四部分·1.3 双指针｜408真题2012年"找公共结点"同源

LC 234回文链表简单高频考点
考点拆解｜数据结构：单链表 ｜ STL：无 ｜ 算法：快慢指针+逆置后半段 ｜ 突破口：找中点→逆后半→逐个比较

判断链表是否回文，要求O(n)、O(1)。（408大题三合一经典）

查看解析

思路推导：快慢指针找中点（偶数取前一个中点）→就地逆置后半段→前半与后半逐结点比较。三个子问题全部来自第四部分模板。

class Solution {
public:
    bool isPalindrome(ListNode* head) {
        if (!head || !head->next) return true;
        ListNode *slow = head, *fast = head;
        while (fast->next && fast->next->next) {   // 中点偏前
            slow = slow->next;
            fast = fast->next->next;
        }
        ListNode *second = slow->next;       // 后半段
        slow->next = NULL;                   // 断开
        ListNode *prev = NULL;
        while (second) {                     // 逆置后半段
            ListNode *nxt = second->next;
            second->next = prev;
            prev = second;
            second = nxt;
        }
        ListNode *p = head, *q = prev;       // 逐个比较
        while (q) {
            if (p->val != q->val) return false;
            p = p->next; q = q->next;
        }
        return true;
    }
};

思路推导：单链表删除需找前驱O(n)→用双向链表：哈希表完成O(1)查找，命中后把结点从链中摘下移到头部；尾部即最久未用，淘汰O(1)。408把该题作为"哈希+链表复合结构"的绝佳案例。

struct Node {
    int key, val;
    Node *prev, *next;
    Node(int k, int v) : key(k), val(v), prev(NULL), next(NULL) {}
};
class LRUCache {
    int cap;
    Node *head, *tail;                        // 哨兵：head侧最近，tail侧最久
    unordered_map<int, Node*> pos;            // key → 结点
    void remove(Node *p) {                    // 摘结点 O(1)
        p->prev->next = p->next;
        p->next->prev = p->prev;
    }
    void pushFront(Node *p) {                 // 头插 O(1)
        p->next = head->next;
        p->prev = head;
        head->next->prev = p;
        head->next = p;
    }
public:
    LRUCache(int capacity) : cap(capacity) {
        head = new Node(0, 0); tail = new Node(0, 0);
        head->next = tail; tail->prev = head;
    }
    int get(int key) {
        unordered_map<int,Node*>::iterator it = pos.find(key);
        if (it == pos.end()) return -1;
        Node *p = it->second;
        remove(p); pushFront(p);              // 移到最近端
        return p->val;
    }
    void put(int key, int value) {
        unordered_map<int,Node*>::iterator it = pos.find(key);
        if (it != pos.end()) {                // 已存在：更新+提前
            Node *p = it->second;
            p->val = value;
            remove(p); pushFront(p);
            return;
        }
        if ((int)pos.size() == cap) {         // 淘汰最久未用
            Node *last = tail->prev;
            remove(last);
            pos.erase(last->key);
            delete last;
        }
        Node *p = new Node(key, value);
        pushFront(p);
        pos[key] = p;
    }
};

查看解析

思路推导：维护"下标对应值递减"的队列：新元素清算队尾所有≤它者（它们不可能再成为最大）、队头滑出窗口则弹出、窗口形成时队头即答案。每元素进出各一次O(n)。

class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        deque<int> q;                         // 存下标，值递减
        vector<int> res;
        for (int i = 0; i < (int)nums.size(); i++) {
            while (!q.empty() && nums[q.back()] <= nums[i])
                q.pop_back();                 // ①队尾清算
            q.push_back(i);                   // ②入队尾
            if (q.front() <= i - k) q.pop_front();   // ③队头过期
            if (i >= k - 1) res.push_back(nums[q.front()]);  // ④答案
        }
        return res;
    }
};

class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();
        for (int i = 0; i < n; i++) {
            while (nums[i] >= 1 && nums[i] <= n
                   && nums[nums[i] - 1] != nums[i]) {   // 没归位就换
                swap(nums[i], nums[nums[i] - 1]);       // x放到下标x-1
            }
        }
        for (int i = 0; i < n; i++)
            if (nums[i] != i + 1) return i + 1;         // 第一个没归位处
        return n + 1;
    }
};

class RandomizedSet {
    vector<int> a;                            // 存元素
    unordered_map<int, int> pos;              // 值 → 下标
public:
    bool insert(int val) {
        if (pos.count(val)) return false;
        pos[val] = a.size();
        a.push_back(val);
        return true;
    }
    bool remove(int val) {
        unordered_map<int,int>::iterator it = pos.find(val);
        if (it == pos.end()) return false;
        int i = it->second, last = a.back();

    int kth(vector<int>& a, int i, vector<int>& b, int j, int k) {
        if (i >= (int)a.size()) return b[j + k - 1];   // a空：b中直接取
        if (j >= (int)b.size()) return a[i + k - 1];
        if (k == 1) return min(a[i], b[j]);

        if (total % 2) return kth(nums1, 0, nums2, 0, total / 2 + 1);
        return (kth(nums1, 0, nums2, 0, total / 2)

                lower_bound(tails.begin(), tails.end(), x);
            if (it == tails.end()) tails.push_back(x);   // 延长LIS
            else *it = x;                    // 缩小结尾，留机会
        }
        return (int)tails.size();
    }
};

返回字符串的最长回文子串。（408串章节的程序化）

查看解析

思路推导：枚举2n−1个回文中心（每个字符=奇中心，每对相邻=偶中心），向两侧扩展到不等，记录最长起止。O(n²)/O(1)——比DP表更省且好写。

class Solution {
    pair<int,int> expand(const string &s, int l, int r) {   // 向外扩展
        while (l >= 0 && r < (int)s.size() && s[l] == s[r]) {
            l--; r++;
        }
        return {l + 1, r - 1};               // 有效回文区间
    }
public:
    string longestPalindrome(string s) {
        int bestL = 0, bestR = 0;
        for (int i = 0; i < (int)s.size(); i++) {
            pair<int,int> a = expand(s, i, i);       // 奇长度中心
            pair<int,int> b = expand(s, i, i + 1);   // 偶长度中心
            if (a.second - a.first > bestR - bestL) { bestL = a.first; bestR = a.second; }
            if (b.second - b.first > bestR - bestL) { bestL = b.first; bestR = b.second; }
        }
        return s.substr(bestL, bestR - bestL + 1);
    }
};
