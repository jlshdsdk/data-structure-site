1.1.1 顺序表定义 顺序表定义

静态分配用定长数组一次到位，动态分配用 malloc/new 按需申请、可扩容，两种定义都要会写。

//1.静态分配的顺序表
#define MaxSize 50                 //定义线性表的最大长度
typedef struct{
    ElemType data[MaxSize];        //顺序表的元素
    int length;                    //顺序表的当前长度
}SqList;                           //顺序表的类型定义

//2.动态分配的顺序表（与静态分配二选一）
#define InitSize 100               //表长度的初始定义
typedef struct{
    ElemType *data;                //指示动态分配数组的指针
    int MaxSize, length;           //数组的最大容量和当前个数
}SeqList;                          //动态分配数组顺序表的类型定义

//C语言的初始动态分配语句为

//C++语言的初始动态分配语句为

1.1.2 顺序表基本操作 顺序表基本操作

静态/动态两种初始化加插入、删除、按值查找。核心是位序 i 与下标 i-1 的换算，以及移动元素的循环方向。

//静态分配的顺序表初始化
//SqList L;                       //声明一个顺序表
void InitList(SqList &L){
    L.length = 0;                  //顺序表初始长度为0
}

//动态分配的顺序表初始化
void InitList(SeqList &L){         //与上面构成重载
    L.data = (ElemType*)malloc(MaxSize*sizeof(ElemType));   //分配存储空间
    L.length = 0;                  //顺序表初始长度为0
    L.MaxSize = InitSize;          //初始存储容量
}

//插入操作：将元素e插入到顺序表L中第i个位置
bool ListInsert(SqList &L, int i, ElemType e){
    if(i<1 || i>L.length+1)        //判断i的范围是否有效
        return false;
    if(L.length >= MaxSize)        //当前存储空间已满，不能插入
        return false;
    for(int j=L.length; j>=i; j--) //将第i个元素及之后的元素后移
        L.data[j] = L.data[j-1];
    L.data[i-1] = e;               //在位置i处放入e
    L.length++;                    //线性表长度加1
    return true;
}

//删除操作：删除顺序表L中第i个位置的元素，用e带回其值
bool ListDelete(SqList &L, int i, ElemType &e){
    if(i<1 || i>L.length)          //判断i的范围是否有效
        return false;
    e = L.data[i-1];               //将被删除的元素赋值给e
    for(int j=i; j<L.length; j++)  //将第i个位置之后的元素前移
        L.data[j-1] = L.data[j];
    L.length--;                    //线性表长度减1
    return true;
}

//按值查找（顺序查找）：查找顺序表中值为e的元素，返回其位序
int LocateElem(SqList L, ElemType e){
    int i;
    for(i=0; i<L.length; i++)
        if(L.data[i] == e)
            return i+1;            //返回其位序i+1
    return 0;                      //退出循环，说明查找失败
}

1.2.1 单链表结点定义 单链表结点定义

结点 = 数据域 + 后继指针；LNode 强调结点，LinkList 强调链表（头指针）。

typedef struct LNode{              //定义单链表结点类型
    ElemType data;                 //数据域，存放一个数据元素
    struct LNode *next;            //指针域，指向下一个结点
}LNode, *LinkList;                 //LNode强调结点，LinkList强调链表（头指针）

1.2.2 单链表基本操作 单链表基本操作

带头/不带头两种初始化，求表长与两种查找，按位序插入删除，头插法（逆置利器）与尾插法建表。改指针的语句顺序是高频考点。

//初始化一个空的单链表（不带头结点）
bool InitList(LinkList &L){
    L = NULL;                            //空表，暂时还没有任何结点
    return true;
}

//初始化一个单链表（带头结点）——王道推荐
bool InitList2(LinkList &L){
    L = (LNode*)malloc(sizeof(LNode));   //创建头结点
    if(L == NULL)                        //内存不足，分配失败
        return false;
    L->next = NULL;                      //头结点之后暂时还没有元素结点
    return true;
}

//求表长（带头结点）
int Length(LinkList L){
    int len = 0;                         //计数变量，初始为0
    LNode *p = L;
    while(p->next != NULL){
        p = p->next;
        len++;                           //每访问一个结点，计数加1
    }
    return len;
}

//按序号查找：取出带头结点单链表L中第i个位置的结点指针
LNode *GetElem(LinkList L, int i){
    if(i < 0)
        return NULL;
    LNode *p = L;                        //指针p指向当前扫描到的结点
    int j = 0;                           //记录当前结点的位序，头结点是第0个结点
    while(p != NULL && j < i){           //循环找到第i个结点
        p = p->next;
        j++;
    }
    return p;                            //返回第i个结点的指针或NULL
}

//按值查找：查找数据域值等于e的结点指针，否则返回NULL
LNode *LocateElem(LinkList L, ElemType e){
    LNode *p = L->next;                  //从第1个结点开始查找
    while(p != NULL && p->data != e)     //找到data域为e的结点即停
        p = p->next;
    return p;                            //找到后返回该结点指针，否则返回NULL
}

//按位序插入（带头结点）：在第i个位置插入元素e
bool ListInsert(LinkList &L, int i, ElemType e){
    if(i < 1)
        return false;
    LNode *p = GetElem(L, i-1);          //查找插入位置的前驱结点
    if(p == NULL)                        //i值不合法
        return false;
    LNode *s = (LNode*)malloc(sizeof(LNode));
    s->data = e;
    s->next = p->next;                   //①新结点先接上p的后继（顺序不能颠倒）
    p->next = s;                         //②再将s连到p之后
    return true;
}

//后插语句：将*s插到*p之后（配合下面的交换数据域可实现"前插"效果，O(1)）
//s->next = p->next;                     //修改指针域，不能颠倒
//p->next = s;
//temp = p->data;                        //交换数据域部分
//p->data = s->data;
//s->data = temp;

//删除*p的后继结点（片段）
//q = p->next;                           //令q指向被删除结点
//p->next = q->next;                     //*p从链中断开
//free(q);                               //释放结点的存储空间

//删除指定结点p本身：与后继结点交换数据域后删后继（O(1)）
bool DeleteNode(LNode *p){
    if(p == NULL || p->next == NULL)     //p是最后一个结点时此法失效，只能O(n)找前驱
        return false;
    LNode *q = p->next;                  //令q指向*p的后继结点
    p->data = p->next->data;             //和后继结点交换数据域
    p->next = q->next;                   //将*q结点从链中"断开"
    free(q);                             //释放后继结点的存储空间
    return true;
}

//头插法建立单链表：从表尾到表头逆向建立，每次均在头结点之后插入元素
LinkList CreateList1(LinkList &L){
    LNode *s; int x;
    L = (LinkList)malloc(sizeof(LNode)); //创建头结点
    L->next = NULL;                      //初始化为空链表
    while(scanf("%d", &x) != EOF){       //循环输入
        s = (LNode*)malloc(sizeof(LNode));   //创建新结点
        s->data = x;
        s->next = L->next;               //将新结点插入表中，L为头指针
        L->next = s;
    }
    return L;
}

//尾插法建立单链表：从表头到表尾正向建立，每次均在表尾插入元素
LinkList CreateList2(LinkList &L){
    int x;                               //设元素类型为整型
    L = (LinkList)malloc(sizeof(LNode)); //创建头结点
    LNode *s, *r = L;                    //r为表尾指针
    while(scanf("%d", &x) != EOF){       //循环输入
        s = (LNode*)malloc(sizeof(LNode));
        s->data = x;
        r->next = s;                     //插入到表尾
        r = s;                           //r指向新的表尾结点
    }
    r->next = NULL;                      //尾结点指针置空
    return L;
}

1.3.1 双链表结点定义 双链表结点定义

在单链表结点基础上增加前驱指针 prior，可双向遍历。

typedef struct DNode{                //定义双链表结点类型
    ElemType data;                   //数据域
    struct DNode *prior, *next;      //前驱和后继指针
}DNode, *DLinkList;

1.3.2 双链表基本操作 双链表基本操作

插入要按①②③④顺序接好四个指针，删除要先接后继再释放，记住"先连后断"。

//初始化双链表（带头结点）
bool InitDLinkList(DLinkList &L){
    L = (DNode*)malloc(sizeof(DNode));   //分配一个头结点
    if(L == NULL)
        return false;                    //内存不足，分配失败
    L->prior = NULL;                     //头结点的prior永远指向NULL
    L->next = NULL;                      //头结点之后暂时还没有结点
    return true;
}

//插入：在p结点之后插入s结点（注意语句顺序！）
bool InsertNextDNode(DNode *p, DNode *s){
    if(p == NULL || s == NULL)           //非法参数
        return false;
    s->next = p->next;                   //①结点*s插入到结点*p之后，先接上p的原后继
    if(p->next != NULL)                  //如果p结点有后继结点
        p->next->prior = s;              //②p的原后继的prior指向s
    s->prior = p;                        //③s的prior指向p
    p->next = s;                         //④p的next指向s（最后才能改！）
    return true;
}

//删除：删除p结点的后继结点q
bool DeleteNextDNode(DNode *p){
    if(p == NULL)
        return false;
    DNode *q = p->next;                  //找到p的后继结点q
    if(q == NULL)                        //p没有后继
        return false;
    p->next = q->next;
    if(q->next != NULL)                  //q结点不是最后一个结点
        q->next->prior = p;
    free(q);                             //释放结点空间
    return true;
}

2.1.1 顺序栈 顺序栈

top 指向栈顶元素：栈空为 -1，栈满为 MaxSize-1；进栈"先加指针再存元素"，出栈相反。

#define MaxSize 50                   //定义栈中元素的最大个数
typedef struct{
    ElemType data[MaxSize];          //存放栈中元素
    int top;                         //栈顶指针
}SqStack;

//(1)初始化
void InitStack(SqStack &S){
    S.top = -1;                      //初始化栈顶指针
}

//(2)判栈空
bool StackEmpty(SqStack S){
    if(S.top == -1)                  //栈空
        return true;
    else                             //不空
        return false;
}

//(3)进栈
bool Push(SqStack &S, ElemType x){
    if(S.top == MaxSize-1)           //栈满，报错
        return false;
    S.data[++S.top] = x;             //指针先加1，再入栈
    return true;
}

//(4)出栈
bool Pop(SqStack &S, ElemType &x){
    if(S.top == -1)                  //栈空，报错
        return false;
    x = S.data[S.top--];             //先出栈，指针再减1
    return true;                     //数据仍残留内存，只是逻辑上被删除
}

//(5)读栈顶元素
bool GetTop(SqStack S, ElemType &x){
    if(S.top == -1)                  //栈空，报错
        return false;
    x = S.data[S.top];               //x记录栈顶元素
    return true;
}

2.1.2 链式栈 链式栈

用链表实现栈，规定所有操作都在表头进行，头指针即栈顶指针，不存在栈满问题。

typedef struct Linknode{             //链栈结点定义
    ElemType data;                   //数据域
    struct Linknode *next;           //指针域
}*LiStack;                           //栈类型定义

//初始化栈（不带头结点）
bool InitStack(LiStack &L){
    L = NULL;                        //空栈
    return true;
}

//判栈空
bool StackEmpty(LiStack L){
    if(L == NULL)
        return true;
    else
        return false;
}

//进栈：相当于对链表头进行"后插"
bool Push(LiStack &L, ElemType x){
    Linknode *s = (Linknode*)malloc(sizeof(Linknode));
    if(s == NULL)                    //内存分配失败
        return false;
    s->data = x;                     //用结点s保存数据元素x
    s->next = L;                     //新结点接在原栈顶之前
    L = s;                           //头指针指向新栈顶
    return true;
}

//出栈：相当于删除链表的第一个结点
bool Pop(LiStack &L, ElemType &x){
    if(L == NULL)                    //栈空，报错
        return false;
    Linknode *p = L;                 //p指向此次出栈的结点
    x = p->data;                     //用x返回栈顶元素
    L = L->next;                     //栈顶指针下移
    free(p);                         //释放结点空间
    return true;
}

2.2.1 顺序队列 顺序队列

循环队列牺牲一个存储单元区分队空与队满：队空 front==rear，队满 (rear+1)%MaxSize==front，指针加1都要取模。

#define MaxSize 50                   //定义队列中元素的最大个数
typedef struct{
    ElemType data[MaxSize];          //用数组存放队列元素
    int front, rear;                 //队头指针和队尾指针
}SqQueue;

//(1)初始化
void InitQueue(SqQueue &Q){
    Q.rear = Q.front = 0;            //初始化队首、队尾指针
}

//(2)判队空
bool isEmpty(SqQueue Q){
    if(Q.rear == Q.front)            //队空条件
        return true;
    else
        return false;
}

//(3)入队（循环队列）
bool EnQueue(SqQueue &Q, ElemType x){
    if((Q.rear+1)%MaxSize == Q.front)    //队满报错（牺牲一个存储单元）
        return false;
    Q.data[Q.rear] = x;              //新元素插入队尾
    Q.rear = (Q.rear+1)%MaxSize;     //队尾指针加1取模
    return true;
}

//(4)出队（循环队列）
bool DeQueue(SqQueue &Q, ElemType &x){
    if(Q.rear == Q.front)            //队空报错
        return false;
    x = Q.data[Q.front];             //x记录队头元素
    Q.front = (Q.front+1)%MaxSize;   //队头指针加1取模
    return true;
}
//队列中元素个数 = (Q.rear - Q.front + MaxSize) % MaxSize

2.2.2 链式队列 链式队列

带头结点的链式队列：front 指头结点；出队时若这是最后一个结点，必须把 rear 拉回头结点。

typedef struct LinkNode{             //链式队列结点
    ElemType data;
    struct LinkNode *next;
}LinkNode;

typedef struct{                      //链式队列
    LinkNode *front, *rear;          //队列的队头和队尾指针
}LinkQueue;

//(1)初始化（带头结点）
void InitQueue(LinkQueue &Q){
    Q.front = Q.rear = (LinkNode*)malloc(sizeof(LinkNode));   //建立头结点
    Q.front->next = NULL;            //初始为空
}

//(2)判队空
bool IsEmpty(LinkQueue Q){
    if(Q.front == Q.rear)            //判空条件
        return true;
    else
        return false;
}

//(3)入队
void EnQueue(LinkQueue &Q, ElemType x){
    LinkNode *s = (LinkNode*)malloc(sizeof(LinkNode));
    s->data = x;
    s->next = NULL;
    Q.rear->next = s;                //插入链尾
    Q.rear = s;                      //修改尾指针
}

//(4)出队
bool DeQueue(LinkQueue &Q, ElemType &x){
    if(Q.front == Q.rear)            //空队
        return false;
    LinkNode *p = Q.front->next;     //p指向此次出队的结点
    x = p->data;
    Q.front->next = p->next;         //摘下结点p
    if(Q.rear == p)                  //若原队列中只有一个结点，删除后变空
        Q.rear = Q.front;            //rear回到头结点（关键！）
    free(p);
    return true;
}

2.3 应用 应用

栈在递归中的作用：函数调用栈保存每层递归的参数、局部变量与返回地址，斐波那契数列是最经典的例子。

//栈在递归中的作用：系统把每层递归的局部变量、参数与返回地址
//压入函数调用栈，返回时逐层出栈恢复现场
//斐波那契数列的递归实现
int F(int n){
    if(n == 0)
        return 0;                    //边界条件
    else if(n == 1)
        return 1;                    //边界条件
    else
        return F(n-1) + F(n-2);      //递归表达式
}

3.1 串的定义与实现 串的定义与实现

串的两种顺序存储：定长顺序表示用静态数组ch[MAXLEN]；堆分配表示按串长动态分配，ch指向串的基地址。

//---------- 定长顺序存储表示 ----------
#define MAXLEN 255                  //预定义最大串长为255
typedef struct{
    char ch[MAXLEN];                //每个分量存储一个字符
    int length;                     //串的实际长度
}SString;

//---------- 堆分配存储表示 ----------
typedef struct{
    char *ch;                       //按串长度分配存储区，ch指向串的基地址
    int length;                     //串的长度
}HString;

3.2.1 简单模式匹配算法 简单模式匹配算法（BF）

朴素模式匹配：匹配失败时主串指针i回退（i=i-j+2）、模式串指针j回到1重新开始，最坏时间复杂度O(mn)。

//简单的模式匹配算法（不使用串的基本操作，直接通过数组下标实现）
int Index(SString S, SString T){
    int i=1, j=1;                   //i为主串指针，j为模式串指针
    while(i<=S.length && j<=T.length){
        if(S.ch[i] == T.ch[j]){
            ++i; ++j;               //继续比较后继字符
        }else{
            i = i-j+2;              //主串指针回退：指向下一个子串的第一个位置
            j = 1;                  //模式串指针后退：重新开始匹配
        }
    }
    if(j > T.length)
        return i-T.length;          //匹配成功：返回子串第一个字符在主串中的位置
    else
        return 0;                   //主串中不存在与T相等的子串
}

3.2.2 KMP算法：求next数组 KMP算法：求next数组

next[1]恒为0；j==0或字符相等时i、j各进一格并置next[i]=j，否则j=next[j]回退。考试以手算为主，此代码了解即可。

//求next数组（考研考手算，代码了解即可）
void get_next(SString T, int next[]){
    int i=1, j=0;
    next[1] = 0;                    //约定：next[1]恒为0
    while(i < T.length){
        if(j==0 || T.ch[i]==T.ch[j]){
            ++i; ++j;
            next[i] = j;            //最长相等前后缀长度 + 1
        }else
            j = next[j];            //前后缀对不上，j回退再试
    }
}

3.2.3 KMP算法：匹配 KMP算法：匹配

KMP匹配：主串指针i不回溯，失配时模式串指针j=next[j]右移；j==0时i++、j++从下一字符重新比较。最坏时间复杂度O(m+n)。

//利用next数组进行匹配（主串指针不回溯）
int Index_KMP(SString S, SString T, int next[]){
    int i=1, j=1;
    while(i<=S.length && j<=T.length){
        if(j==0 || S.ch[i]==T.ch[j]){
            ++i; ++j;               //继续比较后继字符
        }else
            j = next[j];            //模式串向右移动（主串指针i不变）
    }
    if(j > T.length)
        return i-T.length;          //匹配成功
    else
        return 0;
}

3.2.4 由next求nextval数组 由next求nextval数组

若T.ch[next[j]]==T.ch[j]，跳过去必再失配，继承nextval[next[j]]；否则保留next[j]。匹配时把Index_KMP中的next换成nextval即可。

//由next数组求nextval数组（王道标准写法）
void get_nextval(SString T, int next[], int nextval[]){
    nextval[1] = 0;                          //nextval[1]恒为0
    for(int j=2; j<=T.length; j++){
        if(T.ch[next[j]] == T.ch[j])
            nextval[j] = nextval[next[j]];   //跳过去也必失配，继承nextval
        else
            nextval[j] = next[j];            //跳过去可能配上，保留next
    }
}

4.1.2 中序遍历（非递归） 二叉树的中序遍历（非递归）

借助栈：一路向左依次入栈，栈顶出栈并访问，再转向右子树；空间复杂度O(h)。

//二叉树的中序遍历（非递归）
void InOrder2(BiTree T){
    InitStack(S);                       //初始化栈S
    BiTree p = T;                       //p是遍历指针
    while(p || !IsEmpty(S)){            //栈不空或p不空时循环
        if(p){                          //一路向左
            Push(S, p);                 //当前结点入栈
            p = p->lchild;              //左孩子不空，一直往左走
        }else{                          //出栈，访问根结点，遍历右子树
            Pop(S, p);                  //栈顶元素出栈
            visit(p);                   //访问根结点
            p = p->rchild;              //再向右子树走
        }
    }
}

4.1.2 先序遍历（非递归） 二叉树的先序遍历（非递归）

//二叉树的先序遍历（非递归）

        }else{                          //出栈，并转向出栈结点的右子树

4.1.2 层次遍历 二叉树的层次遍历

借助辅助队列：根结点入队，出队访问后将其左、右孩子依次入队，直到队空。

//二叉树的层次遍历
void LevelOrder(BiTree T){
    InitQueue(Q);                       //初始化辅助队列
    BiTree p;
    EnQueue(Q, T);                      //将根结点入队
    while(!IsEmpty(Q)){                 //队列不空则循环
        DeQueue(Q, p);                  //队头结点出队
        visit(p);                       //访问出队结点
        if(p->lchild != NULL)
            EnQueue(Q, p->lchild);      //若左孩子不空，左孩子入队
        if(p->rchild != NULL)
            EnQueue(Q, p->rchild);      //若右孩子不空，右孩子入队
    }
}

4.2.1 线索二叉树的存储结构 线索二叉树的存储结构

在二叉链表基础上增加ltag、rtag：tag==0表示指针指向孩子，tag==1表示指针是线索（利用空链域记录前驱/后继）。

//线索二叉树的存储结构（线索链表）
typedef struct ThreadNode{
    ElemType data;                          //数据元素
    struct ThreadNode *lchild, *rchild;     //左、右孩子指针
    int ltag, rtag;                         //左、右线索标志
}ThreadNode, *ThreadTree;

4.2.2 中序线索二叉树的构造 中序线索二叉树的构造

中序遍历中线索化：左子树空则lchild指向前驱pre；前驱的右子树空则其rchild指向当前结点。主过程最后处理序列的最后一个结点。

//中序遍历对二叉树线索化的递归算法（pre为引用参数，随递归回溯保持最新值）
void InThread(ThreadTree &p, ThreadTree &pre){
    if(p != NULL){
        InThread(p->lchild, pre);               //递归，线索化左子树
        if(p->lchild == NULL){                  //当前结点的左子树为空
            p->lchild = pre;                    //建立当前结点的前驱线索
            p->ltag = 1;
        }
        if(pre != NULL && pre->rchild == NULL){ //前驱结点非空且其右子树为空
            pre->rchild = p;                    //建立前驱结点的后继线索
            pre->rtag = 1;
        }
        pre = p;                                //标记当前结点成为刚刚访问过的结点
        InThread(p->rchild, pre);               //递归，线索化右子树
    }
}

//通过中序遍历建立中序线索二叉树的主过程算法
void CreateInThread(ThreadTree T){
    ThreadTree pre = NULL;
    if(T != NULL){                              //非空二叉树，线索化
        InThread(T, pre);                       //线索化二叉树
        pre->rchild = NULL;                     //处理遍历的最后一个结点
        pre->rtag = 1;
    }
}

4.2.2 中序线索二叉树的遍历 中序线索二叉树的遍历

Firstnode找最左下结点，Nextnode按rtag或右子树最左下找后继；二者串起来即为O(1)空间的非递归中序遍历。

//1.求中序线索二叉树中序序列下的第一个结点
ThreadNode *Firstnode(ThreadNode *p){
    while(p->ltag == 0)
        p = p->lchild;                  //最左下结点（不一定是叶结点）
    return p;
}

//2.求中序线索二叉树中结点p在中序序列下的后继结点
ThreadNode *Nextnode(ThreadNode *p){
    if(p->rtag == 0)
        return Firstnode(p->rchild);    //右子树中最左下结点
    else
        return p->rchild;               //rtag==1，直接返回后继线索
}

//3.求中序线索二叉树中序序列下的最后一个结点
ThreadNode *Lastnode(ThreadNode *p){
    while(p->rtag == 0)
        p = p->rchild;                  //最右下结点（不一定是叶结点）
    return p;
}

//4.求中序线索二叉树中结点p在中序序列下的前驱结点
ThreadNode *Prenode(ThreadNode *p){
    if(p->ltag == 0)
        return Lastnode(p->lchild);     //左子树中最右下结点
    else
        return p->lchild;               //ltag==1，直接返回前驱线索
}

//5.不带头结点的中序线索二叉树的中序遍历
void InOrder(ThreadTree T){
    for(ThreadNode *p = Firstnode(T); p != NULL; p = Nextnode(p))
        visit(p);
}

}

4.4.1 并查集的优化 并查集的优化

Union按结点数小树并大树（根的负值绝对值即集合元素数）；Find找根后把沿途结点直接挂到根下（压缩路径）。

//改进的Union操作：小树合并到大树
void Union(int S[], int Root1, int Root2){
    if(Root1 == Root2)
        return;                         //同一集合，不需要合并
    if(S[Root2] > S[Root1]){            //Root2结点数更少（负值更大）
        S[Root1] += S[Root2];           //累加集合的结点总数
        S[Root2] = Root1;               //小树合并到大树
    }else{                              //Root1结点数更少
        S[Root2] += S[Root1];           //累加集合的结点总数
        S[Root1] = Root2;               //小树合并到大树
    }
}

//改进的Find操作：先找到根结点，再进行压缩路径
int Find(int S[], int x){
    int root = x;
    while(S[root] >= 0)                 //循环找到根
        root = S[root];
    while(x != root){                   //压缩路径
        int t = S[x];                   //t指向x的父结点
        S[x] = root;                    //x直接挂到根结点下面
        x = t;
    }
    return root;                        //返回根结点编号
}

void BFS(ALGraph G, int i){

//Prim算法：从顶点v0出发构造连通网G的最小生成树（邻接矩阵存储，O(|V|^2)）
#define INF 2147483647                      //用int上限表示“无穷”（无边）
void Prim(MGraph G, int v0){
    bool isJoin[MaxVertexNum];              //标记各顶点是否已加入树
    int lowCost[MaxVertexNum];              //各顶点加入树的最低代价
    for(int i=0; i<G.vexnum; i++){          //初始化两个辅助数组
        isJoin[i]=false;
        lowCost[i]=INF;
    }
    lowCost[v0]=0;                          //出发点代价为0，第一轮必选中它
    for(int i=0; i<G.vexnum; i++){          //共n轮，每轮纳入一个新顶点
        int k=-1, min=INF;
        for(int j=0; j<G.vexnum; j++)       //①找未入树且lowCost最低的顶点k
            if(!isJoin[j] && lowCost[j]<min){
                min=lowCost[j];
                k=j;
            }
        isJoin[k]=true;                     //将顶点k纳入生成树
        for(int j=0; j<G.vexnum; j++)       //②用k刷新各未入树顶点的lowCost值
            if(!isJoin[j] && G.Edge[k][j]<lowCost[j])
                lowCost[j]=G.Edge[k][j];
    }
}

//Kruskal算法：边按权值递增依次选边，用并查集判断是否构成回路
typedef struct{ int u, v, w; } Edge;        //边：两端点编号+权值
int fa[MaxVertexNum];                       //并查集父指针
int Find(int x){                            //找所在连通分量的根
    while(fa[x]!=x) x=fa[x];
    return x;
}
void Kruskal(Edge e[], int m, int n){       //m条边，n个顶点
    for(int i=0; i<m-1; i++)                //①将边数组按权值递增排序
        for(int j=0; j<m-1-i; j++)
            if(e[j].w>e[j+1].w){
                Edge t=e[j]; e[j]=e[j+1]; e[j+1]=t;
            }
    for(int i=0; i<n; i++) fa[i]=i;         //②初始化：每个顶点自成一个连通分量
    int cnt=0;                              //已入选边数
    for(int i=0; i<m && cnt<n-1; i++){      //③依次考察当前权值最小的边
        int ru=Find(e[i].u), rv=Find(e[i].v);
        if(ru!=rv){                         //两端点属于不同连通分量，不会成环
            fa[ru]=rv;                      //合并两个连通分量
            cnt++;                          //选入此边，选出n-1条即得最小生成树
        }
    }
}

5.3.2 拓扑排序 拓扑排序

重复输出入度为0的顶点并删其出边（用栈存0入度顶点）；输出顶点数<|V|说明有回路。O(|V|+|E|)。

bool TopologicalSort(Graph G){              //G是以邻接表存储的有向图
    InitStack(S);                           //初始化栈，存储入度为0的顶点
    for(int i=0; i<G.vexnum; i++)
        if(indegree[i]==0)
            Push(S, i);                     //将所有入度为0的顶点进栈
    int count=0;                            //计数，记录当前已经输出的顶点数
    while(!IsEmpty(S)){                     //栈不空，则存在入度为0的顶点
        Pop(S, i);                          //栈顶元素出栈
        print[count++]=i;                   //输出顶点i，记录拓扑序列
        for(p=G.vertices[i].firstarc; p; p=p->nextarc){
            //将所有i指向的顶点的入度减1，并将入度减为0的顶点压入栈S
            v=p->adjvex;
            if(!(--indegree[v]))
                Push(S, v);                 //入度减为0，则入栈
        }
    }//while
    if(count < G.vexnum)
        return false;                       //排序失败：有向图中有回路
    else
        return true;                        //拓扑排序成功
}
//每个顶点进栈出栈一次、每条边被“删除”一次：O(|V|+|E|)

第6章 查找2 段代码

本节代码为《王道数据结构》标准实现，可直接用于期末笔试与408答题；点右上角「复制」即可取用。

6.1.1 顺序查找 顺序查找

0号单元存“哨兵”，从后往前找无需判断越界；查找成功返回下标，失败返回0。

typedef struct{                 //查找表的数据结构（顺序表）
    ElemType *elem;             //元素存储空间基址，建表时按实际长度分配
    int TableLen;               //表的长度
}SSTable;

//顺序查找：数据从下标1开始存，0号位置存“哨兵”
int Search_Seq(SSTable ST, ElemType key){
    int i;
    ST.elem[0] = key;                       //“哨兵”——保证循环一定能终止
    for(i = ST.TableLen; ST.elem[i] != key; --i)  //从后往前找，无需判断越界
        ;
    return i;                               //查找成功返回元素下标；失败返回0
}

6.1.2 折半查找 折半查找

只适用于有序顺序表；mid=(low+high)/2向下取整，失败返回-1。判定树层数=⌈log2(n+1)⌉。

//折半查找：查找成功返回元素下标，失败返回-1
int Binary_Search(SSTable L, ElemType key){
    int low = 0, high = L.TableLen - 1, mid;
    while(low <= high){                     //low>high时查找失败
        mid = (low + high) / 2;             //取中间位置（向下取整）
        if(L.elem[mid] == key)
            return mid;                     //查找成功则返回所在位置
        else if(L.elem[mid] > key)
            high = mid - 1;                 //继续在前半部分查找
        else
            low = mid + 1;                  //继续在后半部分查找
    }
    return -1;                              //查找失败，返回-1
}

第7章 排序8 段代码

本节代码为《王道数据结构》标准实现，可直接用于期末笔试与408答题；点右上角「复制」即可取用。

7.1.1 直接插入排序 直接插入排序

带哨兵A[0]：依次把A[2]~A[n]插入前面已排序序列。最好O(n)，最坏/平均O(n^2)，稳定。

//直接插入排序（带哨兵）
void InsertSort(int A[],int n){
    int i,j;
    for(i=2;i<=n;i++)               //依次将A[2]~A[n]插入到前面已排序序列
        if(A[i]<A[i-1]){            //若A[i]关键码小于其前驱，将A[i]插入有序表
            A[0]=A[i];              //复制为哨兵，A[0]不存放元素
            for(j=i-1;A[0]<A[j];--j) //从后往前查找待插入位置
                A[j+1]=A[j];        //向后挪位
            A[j+1]=A[0];            //复制到插入位置
        }
}

7.1.2 折半插入排序 折半插入排序

用折半查找找插入位置，减少比较次数；移动次数不变，仍O(n^2)，稳定。

//折半插入排序
void InsertSort(int A[],int n){
    int i,j,low,high,mid;
    for(i=2;i<=n;i++){              //依次将A[2]~A[n]插入前面的已排序序列
        A[0]=A[i];                  //将A[i]暂存到A[0]
        low=1; high=i-1;            //设置折半查找的范围
        while(low<=high){           //折半查找（默认递增有序）
            mid=(low+high)/2;       //取中间点
            if(A[mid]>A[0]) high=mid-1; //查找左半子表
            else low=mid+1;         //查找右半子表（相等也右插，保证稳定性）
        }
        for(j=i-1;j>=high+1;--j)    //统一后移元素，空出插入位置
            A[j+1]=A[j];
        A[high+1]=A[0];             //插入操作
    }
}

7.1.3 希尔排序 希尔排序

按增量dk分组做直接插入，增量逐趟减半到1；A[0]只是暂存单元不是哨兵。不稳定。

//希尔排序
void ShellSort(int A[],int n){
    int d,i,j;
    //A[0]只是暂存单元，不是哨兵，当j<=0时，插入位置已到
    for(d=n/2; d>=1; d=d/2)         //步长变化（每次缩小一半）
        for(i=d+1; i<=n; ++i)       //对各子表做直接插入
            if(A[i]<A[i-d]){        //需将A[i]插入有序增量子表
                A[0]=A[i];          //暂存在A[0]
                for(j=i-d; j>0 && A[0]<A[j]; j-=d)
                    A[j+d]=A[j];    //记录后移，查找插入的位置
                A[j+d]=A[0];        //插入
            }
}

7.2.1 冒泡排序 冒泡排序

从后往前两两比较逆序即交换，小元素上浮；flag判断一趟无交换则提前结束。O(n^2)，稳定。

//交换
void swap(int &a,int &b){
    int temp=a;
    a=b;
    b=temp;
}
//冒泡排序
void BubbleSort(int A[],int n){
    for(int i=0; i<n-1; i++){
        bool flag=false;            //表示本趟冒泡是否发生交换的标志
        for(int j=n-1; j>i; j--)    //一趟冒泡过程（从后往前）
            if(A[j-1]>A[j]){        //若为逆序
                swap(A[j-1],A[j]);  //交换
                flag=true;
            }
        if(flag==false)
            return;                 //本趟遍历后没有发生交换，说明表已经有序
    }
}

7.2.2 快速排序 快速排序

Partition以首元素为枢轴划分左右子表，再递归；平均O(nlog2n)，空间O(log2n)，不稳定。

//用第一个元素将待排序序列划分成左右两个部分
int Partition(int A[],int low,int high){
    int pivot=A[low];               //第一个元素作为枢轴
    while(low<high){                //用low、high搜索枢轴的最终位置
        while(low<high && A[high]>=pivot) --high;
        A[low]=A[high];             //比枢轴小的元素移动到左端
        while(low<high && A[low]<=pivot) ++low;
        A[high]=A[low];             //比枢轴大的元素移动到右端
    }
    A[low]=pivot;                   //枢轴元素存放到最终位置
    return low;                     //返回存放枢轴的最终位置
}
//快速排序
void QuickSort(int A[],int low,int high){
    if(low<high){                   //递归跳出的条件
        int pivotpos=Partition(A,low,high); //划分
        QuickSort(A,low,pivotpos-1);        //划分左子表
        QuickSort(A,pivotpos+1,high);       //划分右子表
    }
}

7.3.1 简单选择排序 简单选择排序

每趟从无序区选最小元素与区首交换；比较次数恒为n(n-1)/2，不稳定。

//简单选择排序
void SelectSort(int A[],int n){
    for(int i=0; i<n-1; i++){       //一共进行n-1趟
        int min=i;                  //记录最小元素位置
        for(int j=i+1; j<n; j++)    //在A[i..n-1]中选择最小的元素
            if(A[j]<A[min]) min=j;  //更新最小元素位置
        if(min!=i) swap(A[i],A[min]); //封装的swap()函数共移动元素3次
    }
}

7.3.2 堆排序 堆排序

建立大根堆后每趟堆顶与堆底交换再向下调整；建堆O(n)，排序O(nlog2n)，不稳定。附向上调整AdjustUp。

//建立大根堆
void BuildMaxHeap(int A[],int len){
    for(int i=len/2; i>0; i--)      //从i=⌊n/2⌋到1，反复调整堆
        AdjustDown(A,i,len);
}
//函数AdjustDown将元素k向下进行调整
void AdjustDown(int A[],int k,int len){
    A[0]=A[k];                      //A[0]暂存
    for(int i=2*k; i<=len; i*=2){   //沿key较大的子结点向下筛选
        if(i<len && A[i]<A[i+1])
            i++;                    //取key较大的子结点的下标
        if(A[0]>=A[i]) break;       //筛选结束
        else{
            A[k]=A[i];              //将A[i]调整到双亲结点上
            k=i;                    //修改k值，以便继续向下筛选
        }
    }
    A[k]=A[0];                      //被筛选结点的值放入最终位置
}
//堆排序的完整逻辑
void HeapSort(int A[],int len){
    BuildMaxHeap(A,len);            //初始建堆
    for(int i=len; i>1; i--){       //n-1趟的交换和建堆过程
        swap(A[i],A[1]);            //输出堆顶的元素（和堆底元素交换）
        AdjustDown(A,1,i-1);        //整理，把剩余的i-1个元素整理成堆
    }
}
//向上调整堆的算法（大根堆中插入新元素时，从插入位置k向上调整）
void AdjustUp(int A[],int k){
    A[0]=A[k];                      //A[0]暂存
    int i=k/2;                      //若结点值大于双亲结点，则将双亲结点下调，并继续向上比较
    while(i>0 && A[i]<A[0]){        //双亲结点值小于新结点值
        A[k]=A[i];                  //双亲结点下调
        k=i;
        i=k/2;                      //继续向上比较
    }
    A[k]=A[0];                      //复制到最终位置
}

7.4 归并排序 归并排序

二路归并：Merge将两个有序段合并（取等号保证稳定），MergeSort二分递归。O(nlog2n)，稳定，需O(n)辅助数组。

int *B=(int*)malloc(n*sizeof(int));  //辅助数组B
//A[low..mid]和A[mid+1..high]各自有序，将两个部分归并
void Merge(int A[],int low,int mid,int high){
    int i,j,k;
    for(k=low; k<=high; k++)
        B[k]=A[k];                  //将A中所有元素复制到B中
    for(i=low, j=mid+1, k=i; i<=mid && j<=high; k++){
        if(B[i]<=B[j])
            A[k]=B[i++];            //将较小值复制到A中（取=保证稳定）
        else
            A[k]=B[j++];
    }
    while(i<=mid)  A[k++]=B[i++];   //若第一个表未检测完，复制
    while(j<=high) A[k++]=B[j++];   //若第二个表未检测完，复制
}
void MergeSort(int A[],int low,int high){
    if(low<high){
        int mid=(low+high)/2;       //从中间划分两个子序列
        MergeSort(A,low,mid);       //对左侧子序列进行递归排序
        MergeSort(A,mid+1,high);    //对右侧子序列进行递归排序
        Merge(A,low,mid,high);      //归并
    }
}
